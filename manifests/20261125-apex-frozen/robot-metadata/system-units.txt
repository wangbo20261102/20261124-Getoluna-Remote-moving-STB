# /etc/systemd/system/apex-backend.service
[Unit]
Description=Apex unified backend API and WebSocket service
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=nvidia
Group=nvidia
EnvironmentFile=-/etc/apex/apex.env
WorkingDirectory=/opt/kernelmind/apex
ExecStart=/bin/bash -lc 'source /etc/apex/apex_ros_env.sh; exec ros2 run apex_backend backend'
Restart=on-failure
RestartSec=3
KillSignal=SIGINT
TimeoutStopSec=20

[Install]
WantedBy=multi-user.target
